# Missions 3D Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let students answer by manipulating the 3D scene (6 pilot scenes × 3 levels), with satisfying animated feedback, wired to existing XP / mistakes notebook.

**Architecture:** A new `src/missions/` layer (pure data catalog + pure state machine + pure gesture math, then thin DOM/Three.js adapters) plugs into the existing `MathViewport3D` React component, which already owns the scene params (`a`, `b`, `c`) as React state and pushes them to `MathSceneEngine.updateParams()`. The engine only gains small, game-agnostic hooks (picking by tag, frame callbacks, orbit toggle, screen projection). XP is credited server-side through a new capped endpoint.

**Tech Stack:** React 19, TypeScript, Three.js 0.187 (already installed), `motion` 12 (already installed, import from `motion/react`), Express server (`server.ts`), tests as plain `npx tsx` scripts using `node:assert` (project convention, see `scripts/i18n.check.ts`).

**Spec:** `docs/superpowers/specs/2026-11-22-missions-3d-design.md`

## Spec adjustments (found while reading the code — apply these, they override the spec)

1. **Réfraction ③** becomes a timed series of refracted-angle targets (air → verre) instead of "tap the right material": the scene has a single glass slab.
2. **Error "ghost"** is implemented as a **demonstration**: the scene params animate to the solution, then reset for a retry. Same pedagogical effect, no model cloning.
3. **XP** is credited only by the server for signed-in users (same as exercises today: local mode awards 0 XP). Best stars are stored in `localStorage`.
4. **Élément perturbateur** tag is `step-1` (narrative pillars are indexed from 0).

## Global Constraints

- No new npm dependency.
- Mission logic must never live in `src/three/MathSceneEngine.ts`; the engine only gets the hooks of Task 4.
- Outside mission mode, scenes behave exactly as before (except: mouse wheel scrolls the page; Ctrl + wheel zooms).
- Gestures use Pointer Events only (mouse, touch and pen share one code path). Every gesture has a keyboard equivalent.
- `prefers-reduced-motion: reduce` → no camera zoom, no particles; colour + sound feedback stay.
- XP per mission ≤ 40, recomputed server-side from the catalog (client stars are clamped 1–3).
- All user-facing French strings must be translatable: JSX text or string/template literals (the i18n extractor in `scripts/i18n.ts` picks them up); EN/NL translations are added in Task 8.

// Missions 3D : définitions en données pures (aucune dépendance au moteur ni au DOM).

export type ParamKey = 'a' | 'b' | 'c';
export interface MissionParams {
  a: number;
  b: number;
  c: number;
}

/** drag-x / drag-y : glisser · dial : tourner autour d'un point · tap : toucher un objet étiqueté */
export type GestureKind = 'drag-x' | 'drag-y' | 'dial' | 'tap';

export interface MissionControl {
  param: ParamKey;
  gesture: GestureKind;
  min: number;
  max: number;
  step: number;
  /** pixels de glissé par cran (défaut 24) */
  pxPerStep?: number;
  /** tap : préfixe d'étiquette (« slice- ») ; valeur = numéro de l'étiquette + offset */
  tagPrefix?: string;
  offset?: number;
}

export interface MissionRound {
  prompt: string;
  start: Partial<MissionParams>;
  goal?: (p: MissionParams) => boolean;
  targetTag?: string;
  solution?: Partial<MissionParams>;
}

export interface MissionDef {
  id: string;
  preset: string;
  level: 1 | 2 | 3;
  kind: 'adjust' | 'identify';
  prompt: string;
  hint: string;
  start: Partial<MissionParams>;
  controls?: MissionControl[];
  goal?: (p: MissionParams) => boolean;
  solution?: Partial<MissionParams>;
  targetTag?: string;
  /** niveau 3 : série de manches tirées au hasard */
  rounds?: (rand: () => number) => MissionRound[];
  timeLimitS?: number;
}

export interface MissionResult {
  missionId: string;
  level: 1 | 2 | 3;
  stars: 0 | 1 | 2 | 3;
  timeMs: number;
  prompt: string;
  hint: string;
}

export const r = Math.round;

/** n éléments distincts tirés avec la fonction aléatoire fournie (testable avec une graine). */
export function pickN<T>(items: T[], n: number, rand: () => number): T[] {
  const pool = [...items];
  const out: T[] = [];
  while (out.length < n && pool.length) out.push(pool.splice(Math.floor(rand() * pool.length), 1)[0]);
  return out;
}
```

import type { MissionControl, MissionDef, MissionParams } from '../types.ts';
import { pickN, r } from './helpers.ts';

const toRad = (d: number) => (d * Math.PI) / 180;
const toDeg = (x: number) => (x * 180) / Math.PI;

/** Angle réfracté (°) par Snell-Descartes, null si réflexion totale. Mêmes conventions que la scène : a = i₁, b = n₁, c = n₂. */
export function refractedDeg(i1: number, n1: number, n2: number): number | null {
  const s = (n1 / n2) * Math.sin(toRad(i1));
  return s > 1 ? null : toDeg(Math.asin(s));
}
export const limitDeg = (n1: number, n2: number) => toDeg(Math.asin(n2 / n1));

const dial: MissionControl = { param: 'a', gesture: 'dial', min: 5, max: 80, step: 1 };
const refractsTo = (target: number) => (p: MissionParams) => {
  const i2 = refractedDeg(p.a, p.b, p.c);
  return i2 !== null && Math.abs(i2 - target) <= 1;
};
const incidenceFor = (target: number, n1: number, n2: number) => r(toDeg(Math.asin((n2 / n1) * Math.sin(toRad(target)))));

export const refractionMissions: MissionDef[] = [
  {
    id: 'refraction-1', preset: 'pc_optics_3d', level: 1, kind: 'adjust',
    prompt: 'Règle le laser pour que le rayon réfracté fasse 30° (air → eau)',
    hint: 'n₁·sin(i₁) = n₂·sin(i₂) : sin(i₁) = 1,33 × sin(30°).',
    start: { a: 10, b: 1, c: 1.33 }, solution: { a: incidenceFor(30, 1, 1.33) },
    controls: [dial], goal: refractsTo(30),
  },
  {
    id: 'refraction-2', preset: 'pc_optics_3d', level: 2, kind: 'adjust',
    prompt: 'Trouve l’angle limite eau → air : le rayon doit tout juste se réfléchir totalement',
    hint: 'Angle limite : sin(i_lim) = n₂ / n₁ = 1 / 1,33.',
    start: { a: 20, b: 1.33, c: 1 }, solution: { a: Math.ceil(limitDeg(1.33, 1)) },
    controls: [dial],
    goal: (p) => p.a >= limitDeg(p.b, p.c) && p.a <= limitDeg(p.b, p.c) + 1.5,
  },
  {
    id: 'refraction-3', preset: 'pc_optics_3d', level: 3, kind: 'adjust',
    prompt: 'Réfraction chrono',
    hint: 'Plus le verre est réfringent, plus le rayon se rapproche de la normale.',
    start: { a: 5, b: 1, c: 1.5 }, controls: [dial], timeLimitS: 60,
    rounds: (rand) =>
      pickN([20, 25, 30, 35, 40], 4, rand).map((t) => ({
        prompt: `Obtiens un rayon réfracté de ${t}° (air → verre)`,
        start: { a: 5, b: 1, c: 1.5 },
        solution: { a: incidenceFor(t, 1, 1.5) },
        goal: refractsTo(t),
      })),
  },
];

// Machine à états d'une mission : pure, sans DOM ni Three.js (testable en Node).
import type { MissionDef, MissionRound } from './types.ts';

export type Phase = 'intro' | 'playing' | 'demo' | 'paused' | 'done' | 'aborted';

export interface RunState {
  phase: Phase;
  rounds: MissionRound[];
  round: number;
  roundsWon: number;
  failures: number;
  hintUsed: boolean;
  elapsedMs: number;
  resumedAt: number | null;
  pausedFrom: Phase | null;
}

export type RunEvent =
  | { type: 'begin' | 'solved' | 'wrong' | 'demoDone' | 'pause' | 'resume' | 'timeUp'; now: number }
  | { type: 'hint' }
  | { type: 'abort' };

export const MAX_MISSION_XP = 40;
const LEVEL_XP = [20, 30, 40] as const;
const STAR_FACTOR = [0, 0.5, 0.75, 1] as const;

export function createRun(m: MissionDef, rand: () => number = Math.random): RunState {
  const rounds = m.rounds
    ? m.rounds(rand)
    : [{ prompt: m.prompt, start: m.start, goal: m.goal, targetTag: m.targetTag, solution: m.solution }];
  return { phase: 'intro', rounds, round: 0, roundsWon: 0, failures: 0, hintUsed: false, elapsedMs: 0, resumedAt: null, pausedFrom: null };
}

export const elapsedMs = (s: RunState, now: number) => s.elapsedMs + (s.resumedAt !== null ? now - s.resumedAt : 0);
const freeze = (s: RunState, now: number): RunState => ({ ...s, elapsedMs: elapsedMs(s, now), resumedAt: null });

/** Passe à la manche suivante, ou termine la mission après la dernière. */
const advance = (s: RunState, now: number): RunState =>
  s.round + 1 >= s.rounds.length ? { ...freeze(s, now), phase: 'done' } : { ...s, phase: 'playing', round: s.round + 1 };

export function step(s: RunState, e: RunEvent): RunState {
  switch (e.type) {
    case 'begin':
      return s.phase === 'intro' ? { ...s, phase: 'playing', resumedAt: e.now } : s;
    case 'solved':
      return s.phase === 'playing' ? advance({ ...s, roundsWon: s.roundsWon + 1 }, e.now) : s;
    case 'wrong':
      return s.phase === 'playing' ? { ...s, phase: 'demo', failures: s.failures + 1 } : s;
    case 'demoDone':
      if (s.phase !== 'demo') return s;
      // mission simple : on réessaie la même manche ; chrono : manche suivante
      return s.rounds.length === 1 ? { ...s, phase: 'playing' } : advance(s, e.now);
    case 'hint':
      return { ...s, hintUsed: true };
    case 'pause':
      return s.phase === 'playing' || s.phase === 'demo' ? { ...freeze(s, e.now), phase: 'paused', pausedFrom: s.phase } : s;
    case 'resume':
      return s.phase === 'paused' ? { ...s, phase: s.pausedFrom ?? 'playing', pausedFrom: null, resumedAt: e.now } : s;
    case 'timeUp':
      return s.phase === 'playing' || s.phase === 'demo' ? { ...freeze(s, e.now), phase: 'done' } : s;
    case 'abort':
      return { ...s, phase: 'aborted' };
  }
}

export function computeStars(s: RunState): 0 | 1 | 2 | 3 {
  if (s.roundsWon === 0) return 0;
  if (s.rounds.length > 1) {
    const ratio = s.roundsWon / s.rounds.length;
    return ratio === 1 && !s.hintUsed ? 3 : ratio >= 0.6 ? 2 : 1;
  }
  if (s.failures === 0 && !s.hintUsed) return 3;
  return s.failures <= 1 ? 2 : 1;
}

export function missionXp(level: 1 | 2 | 3, stars: number): number {
  const st = Math.max(0, Math.min(3, Math.round(stars)));
  return Math.min(MAX_MISSION_XP, Math.round(LEVEL_XP[level - 1] * STAR_FACTOR[st]));
}
```

// Gestes de mission : calculs purs + adaptateur Pointer Events (souris, doigt et stylet, même code).
import type { MissionControl, ParamKey } from './types.ts';

export function snap(v: number, min: number, max: number, step: number): number {
  const s = Math.round((v - min) / step) * step + min;
  return Math.min(max, Math.max(min, Number(s.toFixed(6))));
}

export function dragValue(c: MissionControl, startValue: number, dxPx: number, dyPx: number): number {
  const delta = c.gesture === 'drag-x' ? dxPx : -dyPx;
  return snap(startValue + (delta / (c.pxPerStep ?? 24)) * c.step, c.min, c.max, c.step);
}

/** Angle (°) entre la verticale montante passant par le centre et le pointeur : l'angle d'incidence par rapport à la normale. */
export function dialValue(c: MissionControl, cx: number, cy: number, x: number, y: number): number {
  const deg = (Math.atan2(Math.abs(x - cx), cy - y) * 180) / Math.PI;
  return snap(deg, c.min, c.max, c.step);
}

export function tapValue(c: MissionControl, tag: string): number | null {
  if (c.gesture !== 'tap' || !c.tagPrefix || !tag.startsWith(c.tagPrefix)) return null;
  const n = Number(tag.slice(c.tagPrefix.length));
  return Number.isFinite(n) ? snap(n + (c.offset ?? 0), c.min, c.max, c.step) : null;
}

const KEYS: Partial<Record<MissionControl['gesture'], Record<string, number>>> = {
  'drag-x': { ArrowRight: 1, ArrowLeft: -1 },
  dial: { ArrowRight: 1, ArrowLeft: -1 },
  'drag-y': { ArrowUp: 1, ArrowDown: -1 },
};

export function keyValue(c: MissionControl, current: number, key: string): number | null {
  const inc = KEYS[c.gesture]?.[key];
  return inc === undefined ? null : snap(current + inc * c.step, c.min, c.max, c.step);
}

export interface GestureHandlers {
  getValue(p: ParamKey): number;
  onChange(p: ParamKey, v: number): void;
  onTap(tag: string): void;
  onEnd(): void;
  pick(x: number, y: number): string | null;
  dialCenter(): { x: number; y: number };
}

const TAP_SLOP_PX = 6;

/** Branche les gestes sur l'élément ; renvoie la fonction de nettoyage. */
export function attachGestures(el: HTMLElement, controls: MissionControl[], h: GestureHandlers): () => void {
  let drag: { id: number; x: number; y: number; tag: string | null; start: Record<ParamKey, number>; moved: boolean } | null = null;

  const down = (e: PointerEvent) => {
    if (drag) return;
    el.setPointerCapture?.(e.pointerId);
    drag = {
      id: e.pointerId, x: e.clientX, y: e.clientY, tag: h.pick(e.clientX, e.clientY), moved: false,
      start: { a: h.getValue('a'), b: h.getValue('b'), c: h.getValue('c') },
    };
  };

  const move = (e: PointerEvent) => {
    if (!drag || e.pointerId !== drag.id) return;
    const dx = e.clientX - drag.x;
    const dy = e.clientY - drag.y;
    if (!drag.moved && Math.hypot(dx, dy) < TAP_SLOP_PX) return;
    drag.moved = true;
    for (const c of controls) {
      let v: number | null = null;
      if (c.gesture === 'drag-x' || c.gesture === 'drag-y') v = dragValue(c, drag.start[c.param], dx, dy);
      else if (c.gesture === 'dial') {
        const o = h.dialCenter();
        v = dialValue(c, o.x, o.y, e.clientX, e.clientY);
      }
      if (v !== null && v !== h.getValue(c.param)) h.onChange(c.param, v);
    }
  };

  const up = (e: PointerEvent) => {
    if (!drag || e.pointerId !== drag.id) return;
    const d = drag;
    drag = null;
    el.releasePointerCapture?.(e.pointerId);
    if (!d.moved && d.tag) {
      for (const c of controls) {
        const v = tapValue(c, d.tag);
        if (v !== null && v !== h.getValue(c.param)) h.onChange(c.param, v);
      }
      h.onTap(d.tag);
    }
    h.onEnd();
  };

  const key = (e: KeyboardEvent) => {
    for (const c of controls) {
      const v = keyValue(c, h.getValue(c.param), e.key);
      if (v !== null) {
        e.preventDefault();
        if (v !== h.getValue(c.param)) h.onChange(c.param, v);
        h.onEnd();
        return;
      }
    }
  };

  el.addEventListener('pointerdown', down);
  el.addEventListener('pointermove', move);
  el.addEventListener('pointerup', up);
  el.addEventListener('pointercancel', up);
  el.addEventListener('keydown', key);
  return () => {
    el.removeEventListener('pointerdown', down);
    el.removeEventListener('pointermove', move);
    el.removeEventListener('pointerup', up);
    el.removeEventListener('pointercancel', up);
    el.removeEventListener('keydown', key);
  };
}
```

    const hit = this.raycaster.intersectObjects(tagged, false)[0];
    return hit ? String(hit.object.userData.tag) : null;
  }

  public findByTag(tag: string): THREE.Object3D | null {
    let found: THREE.Object3D | null = null;
    this.modelGroup.traverse((o) => {
      if (!found && o.userData.tag === tag) found = o;
    });
    return found;
  }

  public setOrbitEnabled(enabled: boolean) {
    this.controls.enabled = enabled;
  }

  public onFrame(cb: (dt: number) => void): () => void {
    this.frameListeners.add(cb);
    return () => this.frameListeners.delete(cb);
  }

  public projectToScreen(obj: THREE.Object3D): { x: number; y: number } {
    const v = new THREE.Vector3();
    obj.getWorldPosition(v).project(this.camera);
    const rect = this.renderer.domElement.getBoundingClientRect();
    return { x: rect.left + ((v.x + 1) / 2) * rect.width, y: rect.top + ((1 - v.y) / 2) * rect.height };
  }

  public getFxContext(): FxContext {
    return {
      scene: this.scene,
      camera: this.camera,
      modelGroup: this.modelGroup,
      isSoftwareGl: this.isSoftwareGl,
      onFrame: (cb) => this.onFrame(cb),
      findByTag: (tag) => this.findByTag(tag),
    };
  }

// Effets de mission : tweens, particules d'XP, zoom, rebond, surbrillance. Aucune allocation pendant le jeu.
import * as THREE from 'three';
import type { FxContext } from '../three/MathSceneEngine.ts';

export const ease = {
  outCubic: (t: number) => 1 - Math.pow(1 - t, 3),
  outBack: (t: number) => {
    const c1 = 1.70158;
    const c3 = c1 + 1;
    return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
  },
  inOutSine: (t: number) => -(Math.cos(Math.PI * t) - 1) / 2,
};

interface Tween { t: number; dur: number; ease: (t: number) => number; update: (k: number) => void; done?: () => void }
interface Orb { alive: boolean; p: THREE.Vector3; v: THREE.Vector3; life: number }

const ORB_LIFE_S = 0.9;
const GRAVITY = 4.5;

export class MissionFx {
  private tweens = new Set<Tween>();
  private unsub: () => void;
  private orbs: THREE.InstancedMesh;
  private orbState: Orb[];
  private dummy = new THREE.Object3D();
  private highlighted = new Map<THREE.Mesh, THREE.Material | THREE.Material[]>();

  constructor(private ctx: FxContext, private reduced: boolean, accentHex: string) {
    const count = ctx.isSoftwareGl ? 16 : 64;
    this.orbs = new THREE.InstancedMesh(new THREE.SphereGeometry(0.06, 10, 8), new THREE.MeshBasicMaterial({ color: accentHex }), count);
    this.orbs.frustumCulled = false;
    this.orbState = Array.from({ length: count }, () => ({ alive: false, p: new THREE.Vector3(), v: new THREE.Vector3(), life: 0 }));
    this.dummy.scale.setScalar(0);
    this.dummy.updateMatrix();
    for (let i = 0; i < count; i++) this.orbs.setMatrixAt(i, this.dummy.matrix);
    ctx.scene.add(this.orbs);
    this.unsub = ctx.onFrame((dt) => this.update(dt));
  }

  tween(dur: number, update: (k: number) => void, easeFn: (t: number) => number = ease.outCubic, done?: () => void) {
    this.tweens.add({ t: 0, dur, ease: easeFn, update, done });
  }

  burst(origin: { x: number; y: number; z: number }) {
    if (this.reduced) return;
    for (const o of this.orbState) {
      o.alive = true;
      o.life = ORB_LIFE_S;
      o.p.set(origin.x, origin.y, origin.z);
      o.v.set((Math.random() - 0.5) * 3, 2 + Math.random() * 2.5, (Math.random() - 0.5) * 3);
    }
  }

  punchZoom() {
    if (this.reduced) return;
    const cam = this.ctx.camera;
    this.tween(0.5, (k) => {
      cam.zoom = 1 + Math.sin(k * Math.PI) * 0.07;
      cam.updateProjectionMatrix();
    }, ease.inOutSine);
  }

  bounce(tag: string) {
    const obj = this.ctx.findByTag(tag) ?? this.ctx.modelGroup;
    const base = obj.scale.clone();
    this.tween(0.6, (k) => obj.scale.copy(base).multiplyScalar(1 + Math.sin(k * Math.PI) * 0.18), ease.outCubic, () => obj.scale.copy(base));
  }

  highlight(tag: string, hex: string) {
    const obj = this.ctx.findByTag(tag);
    if (!(obj instanceof THREE.Mesh) || this.highlighted.has(obj) || Array.isArray(obj.material)) return;
    if (!('emissive' in obj.material)) return;
    this.highlighted.set(obj, obj.material);
    const m = (obj.material as THREE.MeshStandardMaterial).clone();
    m.emissive = new THREE.Color(hex);
    m.emissiveIntensity = 0.7;
    obj.material = m; // clone : les matériaux du moteur sont partagés entre objets
  }

  clearHighlights() {
    for (const [mesh, original] of this.highlighted) {
      (mesh.material as THREE.Material).dispose();
      mesh.material = original;
    }
    this.highlighted.clear();
  }

  aliveOrbs(): number {
    return this.orbState.filter((o) => o.alive).length;
  }

  private update(dt: number) {
    for (const tw of this.tweens) {
      tw.t = Math.min(1, tw.t + dt / tw.dur);
      tw.update(tw.ease(tw.t));
      if (tw.t >= 1) {
        this.tweens.delete(tw);
        tw.done?.();
      }
    }
    let dirty = false;
    for (let i = 0; i < this.orbState.length; i++) {
      const o = this.orbState[i];
      if (!o.alive) continue;
      dirty = true;
      o.v.y -= GRAVITY * dt;
      o.p.addScaledVector(o.v, dt);
      o.life -= dt;
      if (o.life <= 0) o.alive = false;
      this.dummy.position.copy(o.p);
      this.dummy.scale.setScalar(o.alive ? o.life / ORB_LIFE_S : 0);
      this.dummy.updateMatrix();
      this.orbs.setMatrixAt(i, this.dummy.matrix);
    }
    if (dirty) this.orbs.instanceMatrix.needsUpdate = true;
  }

  dispose() {
    this.unsub();
    this.tweens.clear();
    this.clearHighlights();
    this.ctx.scene.remove(this.orbs);
    this.orbs.geometry.dispose();
    (this.orbs.material as THREE.Material).dispose();
    this.ctx.camera.zoom = 1;
    this.ctx.camera.updateProjectionMatrix();
  }
}

  } catch {
    // stockage indisponible : le choix vaut pour la session
  }

function beep(freq: number, dur: number, type: OscillatorType = 'sine', gain = 0.04, delay = 0) {
  if (!soundOn || typeof window === 'undefined' || !('AudioContext' in window)) return;
  ctx ??= new AudioContext();
  if (ctx.state === 'suspended') void ctx.resume();
  const t0 = ctx.currentTime + delay;
  const osc = ctx.createOscillator();
  const g = ctx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  g.gain.setValueAtTime(gain, t0);
  g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
  osc.connect(g).connect(ctx.destination);
  osc.start(t0);
  osc.stop(t0 + dur);
}

// Validation d'un résultat de mission côté serveur : les XP viennent du catalogue, jamais du client.
import { findMission } from './catalog/index.ts';
import { missionXp } from './runner.ts';

const MIN_HUMAN_MS = 1500; // ponytail: plancher anti-robot simple ; plafonner par mission et par jour si un abus apparaît

export function validateMissionCompletion(body: unknown): { ok: true; xp: number } | { ok: false } {
  if (!body || typeof body !== 'object') return { ok: false };
  const { missionId, stars, timeMs } = body as Record<string, unknown>;
  const mission = typeof missionId === 'string' ? findMission(missionId) : null;
  if (!mission || !Number.isInteger(stars) || (stars as number) < 1 || (stars as number) > 3) return { ok: false };
  if (typeof timeMs !== 'number' || !(timeMs >= MIN_HUMAN_MS)) return { ok: false };
  return { ok: true, xp: missionXp(mission.level, stars as number) };
}
```
